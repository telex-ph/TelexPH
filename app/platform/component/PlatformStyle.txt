/* ─── Perspective Fan Slider (Section 1) ─────────────────────────────────── */
function HorizontalPhotoMosaic() {
  const containerRef = useRef<HTMLDivElement | null>(null);
  const rafRef       = useRef<number>(0);
  const offsetRef    = useRef(0);
  const isDragging   = useRef(false);
  const dragStartX   = useRef(0);
  const dragStartOff = useRef(0);
  const velRef       = useRef(0);
  const lastX        = useRef(0);
  const autoRef      = useRef(true);

  const isMobile  = typeof window !== "undefined" && window.innerWidth < 640;
  const CARD_W    = isMobile ? 80 : 130;
  const CARD_H    = isMobile ? 120 : 200;
  const STEP      = CARD_W + 16;
  const COUNT     = ALL_PLATFORMS.length;
  const AUTO_SPEED = 0.012;
  const VISIBLE   = isMobile ? 4 : 5;

  function getStyle(dist: number): React.CSSProperties {
    const absDist = Math.abs(dist);
    const scale   = Math.max(0.62, 1 - absDist * 0.085);
    const rotateY = dist * 6;
    const tx      = dist * STEP * 0.88;
    const ty      = absDist * absDist * 3.5;
    const zIndex  = Math.round(100 - absDist * 10);
    const opacity = Math.max(0.35, 1 - absDist * 0.13);
    return {
      position: "absolute", width: CARD_W, height: CARD_H,
      transform: `translateX(${tx}px) translateY(${ty}px) scale(${scale}) rotateY(${rotateY}deg)`,
      zIndex, opacity,
      transition: isDragging.current ? "none" : "transform 0.05s linear, opacity 0.05s linear",
      borderRadius: 18, overflow: "hidden",
      boxShadow: "0 8px 32px rgba(0,0,0,0.18)",
      background: "#ccc", cursor: "grab", flexShrink: 0, willChange: "transform",
    };
  }

  const [renderOffset, setRenderOffset] = useState(0);

  useEffect(() => {
    let lastTime = 0;
    function animate(ts: number) {
      const dt = Math.min(ts - lastTime, 32);
      lastTime = ts;
      if (!isDragging.current) {
        if (autoRef.current) {
          offsetRef.current += AUTO_SPEED * (dt / 16);
        } else {
          velRef.current *= 0.94;
          offsetRef.current += velRef.current;
          if (Math.abs(velRef.current) < 0.001) autoRef.current = true;
        }
        offsetRef.current = ((offsetRef.current % COUNT) + COUNT) % COUNT;
      }
      setRenderOffset(offsetRef.current);
      rafRef.current = requestAnimationFrame(animate);
    }
    rafRef.current = requestAnimationFrame(animate);
    return () => cancelAnimationFrame(rafRef.current);
  }, []);

  function onPointerDown(e: React.PointerEvent) {
    isDragging.current = true; autoRef.current = false; velRef.current = 0;
    dragStartX.current = e.clientX; dragStartOff.current = offsetRef.current;
    lastX.current = e.clientX;
    (e.currentTarget as HTMLElement).setPointerCapture(e.pointerId);
  }
  function onPointerMove(e: React.PointerEvent) {
    if (!isDragging.current) return;
    const dx = e.clientX - dragStartX.current;
    velRef.current = (lastX.current - e.clientX) / STEP;
    lastX.current = e.clientX;
    offsetRef.current = ((dragStartOff.current + dx / STEP * -1) % COUNT + COUNT) % COUNT;
    setRenderOffset(offsetRef.current);
  }
  function onPointerUp() { isDragging.current = false; }

  const centerIdx = Math.round(renderOffset);
  const cards = [];
  for (let d = -VISIBLE; d <= VISIBLE; d++) {
    const idx  = ((centerIdx + d) % COUNT + COUNT) % COUNT;
    const frac = renderOffset - centerIdx;
    const dist = d - frac;
    cards.push({ idx, dist });
  }

  return (
    <div ref={containerRef}
      style={{
        position: "relative", width: "100%", height: CARD_H + 40,
        overflow: "hidden", perspective: 900, perspectiveOrigin: "50% 100%",
        userSelect: "none", touchAction: "none",
      }}
      onPointerDown={onPointerDown} onPointerMove={onPointerMove}
      onPointerUp={onPointerUp} onPointerLeave={onPointerUp}
    >
      <div style={{ position:"absolute",left:0,top:0,bottom:0,width:isMobile?40:120,background:"linear-gradient(to right, #f7f7f7 35%, transparent)",zIndex:200,pointerEvents:"none" }} />
      <div style={{ position:"absolute",right:0,top:0,bottom:0,width:isMobile?40:120,background:"linear-gradient(to left, #f7f7f7 35%, transparent)",zIndex:200,pointerEvents:"none" }} />
      <div style={{ position:"absolute",left:"50%",top:16,transform:"translateX(-50%)",width:0,height:CARD_H }}>
        {cards.map(({ idx, dist }) => {
          const name  = ALL_PLATFORMS[idx];
          const photo = PHOTOS[idx % PHOTOS.length];
          return (
            <div key={`${idx}-${dist.toFixed(1)}`} style={getStyle(dist)}>
              <img src={photo} alt={name}
                style={{ width:"100%",height:"100%",objectFit:"cover",display:"block",pointerEvents:"none" }}

/* ─── Section 3: Image Slider with auto-advance + free swipe ─────────────── */
function PlatformImageSlider({ items, selectedCountry }: { items: Platform[]; selectedCountry: string }) {
  const trackRef      = useRef<HTMLDivElement>(null);
  const rafRef        = useRef<number>(0);
  const autoTimerRef  = useRef<ReturnType<typeof setTimeout> | null>(null);

  /* physics refs */
  const isDragging    = useRef(false);
  const startX        = useRef(0);
  const startScroll   = useRef(0);
  const velRef        = useRef(0);
  const lastX         = useRef(0);
  const lastT         = useRef(0);

  const [activeIdx, setActiveIdx] = useState(0);

  const CARD_W    = 320;
  const GAP       = 20;
  const STRIDE    = CARD_W + GAP;
  const AUTO_MS   = 5000;

  /* Reset on country change */
  useEffect(() => {
    setActiveIdx(0);
    if (trackRef.current) trackRef.current.scrollLeft = 0;
    velRef.current = 0;
  }, [selectedCountry]);

  /* Scroll to a given index with smooth animation */
  const scrollToIdx = (idx: number) => {
    const clamped = Math.max(0, Math.min(idx, items.length - 1));
    setActiveIdx(clamped);
    if (!trackRef.current) return;
    const target = clamped * STRIDE;
    const start  = trackRef.current.scrollLeft;
    const dist   = target - start;
    const dur    = 420;
    const t0     = performance.now();
    cancelAnimationFrame(rafRef.current);
    function ease(t: number) { return t < 0.5 ? 2*t*t : -1+(4-2*t)*t; }
    function step(now: number) {
      const pct = Math.min((now - t0) / dur, 1);
      if (trackRef.current) trackRef.current.scrollLeft = start + dist * ease(pct);
      if (pct < 1) rafRef.current = requestAnimationFrame(step);
    }
    rafRef.current = requestAnimationFrame(step);
  };

  /* Schedule next auto-advance */
  const scheduleAuto = () => {
    if (autoTimerRef.current) clearTimeout(autoTimerRef.current);
    autoTimerRef.current = setTimeout(() => {
      setActiveIdx(prev => {
        const next = (prev + 1) % items.length;
        scrollToIdx(next);
        return next;
      });
    }, AUTO_MS);
  };

  useEffect(() => {
    scheduleAuto();
    return () => { if (autoTimerRef.current) clearTimeout(autoTimerRef.current); };
  }, [items.length, selectedCountry]);

  /* Pointer / drag handlers */
  const onPointerDown = (e: React.PointerEvent) => {
    cancelAnimationFrame(rafRef.current);
    if (autoTimerRef.current) clearTimeout(autoTimerRef.current);
    isDragging.current  = true;
    startX.current      = e.clientX;
    startScroll.current = trackRef.current?.scrollLeft ?? 0;
    velRef.current      = 0;
    lastX.current       = e.clientX;
    lastT.current       = e.timeStamp;
    (e.currentTarget as HTMLElement).setPointerCapture(e.pointerId);
  };

  const onPointerMove = (e: React.PointerEvent) => {
    if (!isDragging.current || !trackRef.current) return;
    const dx = e.clientX - startX.current;
    trackRef.current.scrollLeft = startScroll.current - dx;
    const dt = e.timeStamp - lastT.current;
    if (dt > 0) velRef.current = (lastX.current - e.clientX) / dt;
    lastX.current = e.clientX;
    lastT.current = e.timeStamp;
  };

  const onPointerUp = (e: React.PointerEvent) => {
    if (!isDragging.current || !trackRef.current) return;
    isDragging.current = false;

    /* momentum glide */
    let vel = velRef.current * 16;
    cancelAnimationFrame(rafRef.current);
    function glide() {
      if (!trackRef.current) return;
      trackRef.current.scrollLeft += vel;
      vel *= 0.88;
      if (Math.abs(vel) > 0.5) {
        rafRef.current = requestAnimationFrame(glide);
      } else {
        /* snap to nearest */
        const nearest = Math.round(trackRef.current.scrollLeft / STRIDE);
        scrollToIdx(Math.max(0, Math.min(nearest, items.length - 1)));
        scheduleAuto();
      }
    }
    rafRef.current = requestAnimationFrame(glide);
  };

  /* Track scroll → sync activeIdx indicator */
  const onScroll = () => {
    if (!trackRef.current || isDragging.current) return;
    const nearest = Math.round(trackRef.current.scrollLeft / STRIDE);
    setActiveIdx(Math.max(0, Math.min(nearest, items.length - 1)));
  };

  const activePlatform = items[activeIdx];
  const description    = platformDescriptions[activePlatform?.name] ?? platformDescriptions["ALL"];

  return (
    <div className="section3-layout flex flex-col lg:flex-row items-stretch w-full" style={{ minHeight: 340 }}>

      {/* LEFT — description */}
      <div
        className="section3-left flex flex-col justify-center px-8 py-10 lg:py-0 flex-shrink-0"
        style={{ width: "clamp(280px, 38%, 440px)" }}
      >
        <h2
          className="txt-fade txt-visible uppercase leading-none mb-5"
          style={{
            fontFamily: TYPOGRAPHY.heading.fontFamily,
            fontWeight: TYPOGRAPHY.heading.fontWeight,
            fontSize: "clamp(24px, 3vw, 42px)",
            color: COLORS.white,
            letterSpacing: "-0.02em",
          }}
        >
          {activePlatform?.name ?? "Featured Platforms"}
        </h2>
        <p
          className="txt-fade txt-visible mb-6 leading-[1.8]"
          style={{ fontFamily: FONTS.rubik, fontSize: 14, color: getColorWithOpacity("white", 0.6), transitionDelay: "0.05s" }}
        >
          {description}
        </p>
        <div className="txt-fade txt-visible flex items-center gap-3" style={{ transitionDelay: "0.1s" }}>
          <span
            className="text-[11px] font-bold tracking-[2px] uppercase bg-white/10 px-4 py-1.5 rounded-full border border-white/15"
            style={{ fontFamily: FONTS.openSans, color: "rgba(255,255,255,0.7)" }}
          >
            {activePlatform?.country ?? ""}
          </span>
          <span style={{ fontFamily: FONTS.rubik, fontSize: 12, color: "rgba(255,255,255,0.35)" }}>
            Marketplace Partner
          </span>
        </div>
      </div>

      {/* RIGHT — scrollable image cards */}
      <div className="section3-right flex-1 relative overflow-hidden" style={{ minHeight: 340 }}>
        {/* only right fade — active card should show fully on left */}
        <div style={{ position:"absolute",right:0,top:0,bottom:0,width:48,background:"linear-gradient(to left,#282828 30%,transparent)",zIndex:10,pointerEvents:"none" }} />

        <div
          ref={trackRef}
          onPointerDown={onPointerDown}
          onPointerMove={onPointerMove}
          onPointerUp={onPointerUp}
          onPointerLeave={onPointerUp}
          onScroll={onScroll}
          className="flex h-full overflow-x-auto"
          style={{
            gap: GAP,
            scrollbarWidth: "none",
            cursor: "grab",
            userSelect: "none",
            touchAction: "pan-y",
            paddingRight: 32,
            scrollSnapType: "none",
          }}
        >
          {items.map((p, i) => (
            <div
              key={i}
              onClick={() => { scrollToIdx(i); scheduleAuto(); }}
              className="flex-shrink-0 relative rounded-2xl overflow-hidden group/card"
              style={{
                width: CARD_W,
                height: 320,
                transform: i === activeIdx ? "scale(1)" : "scale(0.88)",
                opacity: i === activeIdx ? 1 : 0.45,
                boxShadow: i === activeIdx
                  ? "0 20px 60px rgba(0,0,0,0.6)"
                  : "0 4px 16px rgba(0,0,0,0.2)",
                transition: "transform 0.4s ease, opacity 0.4s ease, box-shadow 0.4s ease",
                cursor: "pointer",
              }}
            >
              <img
                src={p.image} alt={p.name} draggable={false}
                className="absolute inset-0 w-full h-full object-cover transition-transform duration-700 group-hover/card:scale-105"
              />
              <div className="absolute inset-0 bg-gradient-to-t from-black/60 via-black/15 to-transparent" />

              {/* labels */}
              <div className="absolute bottom-0 left-0 right-0 p-7 text-white">
                <span
                  className="text-[11px] font-bold tracking-[2px] uppercase bg-white/20 px-4 py-1 rounded-full backdrop-blur-md mb-3 inline-block"
                  style={{ fontFamily: FONTS.openSans, fontWeight: FONT_WEIGHTS.bold }}
                >
                  {p.country}
                </span>
                <div
                  className="leading-none mb-1.5"
                  style={{
                    fontFamily: TYPOGRAPHY.heading.fontFamily,
                    fontWeight: TYPOGRAPHY.heading.fontWeight,
                    fontSize: "clamp(26px, 3vw, 40px)",
                    letterSpacing: "-0.02em",
                    color: COLORS.white,
                  }}
                >
                  {p.name}
                </div>
                <p style={{ fontFamily: FONTS.rubik, fontSize: 13, color: "rgba(255,255,255,0.65)" }}>
                  Marketplace Partner
                </p>
              </div>

              {/* arrow badge */}
              <div className="absolute top-5 right-5 w-10 h-10 rounded-full border-2 border-white/50 flex items-center justify-center text-white/60 group-hover/card:bg-white group-hover/card:text-black transition-all duration-300">
                →
              </div>
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}
